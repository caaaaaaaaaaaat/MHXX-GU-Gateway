cmake_minimum_required(VERSION 3.20)
project(crossplay_dll_probe LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This infrastructure targets native Windows x64 only")
endif()
add_library(dll_probe SHARED src/dll_probe.cpp)
set_target_properties(dll_probe PROPERTIES PREFIX "" CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_include_directories(dll_probe PUBLIC include)
target_compile_definitions(dll_probe PRIVATE DP_BUILD_DLL _WIN32_WINNT=0x0601)
add_executable(loader_host tests/loader_host.cpp)
target_include_directories(loader_host PRIVATE include)
target_compile_definitions(loader_host PRIVATE _WIN32_WINNT=0x0601)
set_target_properties(loader_host PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
add_library(header_c_smoke OBJECT tests/header_c_smoke.c)
target_include_directories(header_c_smoke PRIVATE include)
set_target_properties(header_c_smoke PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MINGW)
    foreach(target dll_probe loader_host)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endforeach()
    target_link_options(dll_probe PRIVATE -Wl,--exclude-all-symbols)
    target_link_options(loader_host PRIVATE -municode)
elseif(MSVC)
    target_compile_options(dll_probe PRIVATE /W4 /EHsc)
    target_compile_options(loader_host PRIVATE /W4 /EHsc)
endif()
enable_testing()
add_test(NAME owned_loader_lifecycle COMMAND loader_host $<TARGET_FILE:dll_probe>)
