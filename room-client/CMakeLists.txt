cmake_minimum_required(VERSION 3.20)
project(room_client_m1 VERSION 0.1.0 LANGUAGES CXX)
# No FetchContent, emulator source include, ENet shim, or network dependency.
add_library(room_protocol STATIC src/protocol.cpp src/session.cpp)
target_include_directories(room_protocol PUBLIC include)
target_compile_features(room_protocol PUBLIC cxx_std_20)
set_target_properties(room_protocol PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(room_protocol PRIVATE /W4 /permissive-)
else()
    target_compile_options(room_protocol PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()
add_executable(room-fixture src/main.cpp)
target_link_libraries(room-fixture PRIVATE room_protocol)
if(MINGW)
    target_compile_options(room_protocol PRIVATE -pipe)
    target_compile_options(room-fixture PRIVATE -pipe)
    target_link_options(room-fixture PRIVATE -Wl,--no-insert-timestamp)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(room-protocol-tests tests/test_protocol.cpp)
    target_link_libraries(room-protocol-tests PRIVATE room_protocol)
    if(MINGW)
        target_compile_options(room-protocol-tests PRIVATE -pipe)
        target_link_options(room-protocol-tests PRIVATE -Wl,--no-insert-timestamp)
    endif()
    add_test(NAME room-protocol-tests COMMAND room-protocol-tests)
    add_test(NAME fixture-decode COMMAND room-fixture decode s2c 02c0a80102)
    add_test(NAME fixture-emit-join COMMAND room-fixture emit-join Test)
    add_test(NAME fixture-replay COMMAND room-fixture replay Test fixtures/join-success.hex)
    set_tests_properties(fixture-replay PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME fixture-reject-truncated COMMAND room-fixture decode s2c 02c0a8)
    set_tests_properties(fixture-reject-truncated PROPERTIES WILL_FAIL TRUE)
endif()
